Python 
import scenario

# Initialize client with your API key
client = scenario.Client(
    api_key="sc_your_api_key_here"
)

# Generate assets using your custom model
result = client.generate(
    model_id="your-custom-model-id",
    prompt="Fantasy warrior, epic armor, game art",
    num_images=4,
    width=1024,
    height=1024,
    style_strength=0.85
)

# Save all generated images
for i, img in enumerate(result.images):
    img.save(f"warrior_{i}.png")

print(f"✓ Generated {len(result.images)} assets")
